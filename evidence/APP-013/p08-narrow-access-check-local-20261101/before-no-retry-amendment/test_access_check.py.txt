class FakeIO:
    proof=True
    def __init__(self,case,cap):self.case=case;self.cap=cap;self.calls=[];self.closed=False
    def cli(self,i,deadline):
        self.calls.append(('cli',i))
        if self.case.break_at==len(self.calls):raise a.Stop('INTERRUPTED')
        q=self.case
        values=[a.OWNER,'(unset)','','(unset)',json.dumps({'projectId':'signmons',
          'projectNumber':q.number,'lifecycleState':'ACTIVE'}),*q.role_json,'SYNTHETIC_BEARER_NEVER_PERSIST']
        return q.cli_override.get(i,values[i]).encode() if isinstance(q.cli_override.get(i,values[i]),str) else q.cli_override[i]
    def https(self,i,number,token,deadline):
        self.calls.append(('https',i))
        if self.case.break_at==len(self.calls):raise a.Stop('INTERRUPTED')
        self.case.assertEqual(number,self.case.number)
        self.case.assertEqual(token,'SYNTHETIC_BEARER_NEVER_PERSIST')
        value={'permissions':[a.PROJECT_PERMISSIONS,a.IDENTITY_PERMISSIONS][i]} if i<2 else {
          'name':a.http_contract(number)[2]['url'].split('/v1/')[1],'state':self.case.service}
        return self.case.http_override.get(i,{'status':200,'body':json.dumps(value).encode()})
    def close(self):self.closed=True;return not self.case.close_failure
